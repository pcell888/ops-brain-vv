## 二、优化原则（保持不变）

1. **接口契约不变**：不改变任何 API 路由、请求参数、响应结构
2. **轻量化架构**：小项目不用 DDD / 六边形，但分层清晰、职责清晰是必要约束
3. **渐进式重构**：按优先级分阶段执行，每阶段可独立验证
4. **零风险先行**：P0 纯机械重构，只移不删，只增不改
5. **先验证后推进**：每阶段完成后运行测试，通过后再进入下一阶段

#### 4.1.1 新建 `src/core/phases.py` — 统一阶段定义

消除 3 处重复的 `_PHASE_NAME` + `_PHASE_OVERALL_RANGE` + `_calc_overall_progress`。

```python
"""阶段定义与进度计算 — 全流程单一信源。"""

from __future__ import annotations

PHASE_DIAGNOSIS = "diagnosis"
PHASE_ADOPTION = "adoption"
PHASE_EXECUTION = "execution"
PHASE_TRACKING = "tracking"
PHASE_COMPLETED = "completed"

_PHASE_NAME = {
    PHASE_DIAGNOSIS: "诊断",
    PHASE_ADOPTION: "采纳",
    PHASE_EXECUTION: "执行",
    PHASE_TRACKING: "追踪",
    PHASE_COMPLETED: "已完成",
}

# (min_percent, max_percent) 区间
_PHASE_OVERALL_RANGE = {
    PHASE_DIAGNOSIS: (0, 60),
    PHASE_ADOPTION: (60, 75),
    PHASE_EXECUTION: (75, 90),
    PHASE_TRACKING: (90, 99),
    PHASE_COMPLETED: (100, 100),
}

WORKFLOW_NODES = frozenset({
    "collect_data", "diagnose", "generate_solutions",
    "wait_adoption", "execute_plans", "track_effects",
})


def calc_overall_progress(phase: str, phase_progress: int) -> int:
    p = max(0, min(100, int(phase_progress)))
    lo, hi = _PHASE_OVERALL_RANGE.get(phase, (0, 100))
    if lo == hi:
        return lo
    return int(round(lo + (hi - lo) * (p / 100.0)))


def phase_name(phase: str) -> str:
    return _PHASE_NAME.get(phase, phase)


def infer_next_phase(phase: str, status: str) -> str | None:
    if status in ("failed", "not_found"):
        return None
    order = [PHASE_DIAGNOSIS, PHASE_ADOPTION, PHASE_EXECUTION, PHASE_TRACKING, PHASE_COMPLETED]
    try:
        idx = order.index(phase)
        return order[idx + 1] if idx + 1 < len(order) else None
    except ValueError:
        return None
```

#### 4.1.2 新建 `src/core/progress_utils.py` — 提取真正重复的进度小工具

**不**强行统一"进度提取大函数"，而是提取两处真正重复的小模式：

```python
"""进度相关小工具 — 提取真正重复的代码模式。"""

from __future__ import annotations

from src.runtime.running_tasks import running_tasks


def is_thread_running(thread_id: str) -> bool:
    """判断 thread 是否在运行（本地 Task + Redis 双重检查）。

    注意：这是同步函数，只检查本地 Task 状态。如需完整检查（含 Redis），请用：
        (task is not None and not task.done()) or await running_tasks.is_running(thread_id)
    """
    task = running_tasks.get(thread_id)
    return task is not None and not task.done()


async def is_thread_running_full(thread_id: str) -> bool:
    """完整判断 thread 是否在运行（本地 Task + Redis）。"""
    task = running_tasks.get(thread_id)
    return (task is not None and not task.done()) or await running_tasks.is_running(thread_id)


def safe_percent(value: Any) -> int:
    """安全地将任意值转换为 0-100 的整数 percent。"""
    try:
        return max(0, min(100, int(float(value))))
    except (TypeError, ValueError):
        return 0
```

#### 4.1.3 新建 `src/core/json_utils.py` — 统一 JSON 工具

```python
"""JSON 解析工具 — 统一 llm_caller 与 agent/tools 的实现。"""

from __future__ import annotations

import json
from typing import Any


def strip_json_fence(text: str) -> str:
    """去除 markdown JSON 围栏（```json ... ```）。"""
    s = text.strip()
    if not s.startswith("```"):
        return s
    lines = s.split("\n")
    if lines and lines[0].startswith("```"):
        lines = lines[1:]
    if lines and lines[-1].strip() == "```":
        lines = lines[:-1]
    return "\n".join(lines).strip()


def parse_json_fence(text: str) -> Any | None:
    """尝试解析含 markdown 围栏的 JSON 字符串。"""
    body = strip_json_fence(text)
    if not body:
        return None
    try:
        return json.loads(body)
    except json.JSONDecodeError:
        return None
```

"""统一 Redis 异步客户端 — 单例 + 生命周期管理。"""

from __future__ import annotations

import logging

import redis.asyncio as aioredis

from src.core.config import get_settings

logger = logging.getLogger(__name__)

_redis: aioredis.Redis | None = None


async def get_redis() -> aioredis.Redis:
    """获取或创建 Redis 连接（单例）。"""
    global _redis
    if _redis is None:
        _redis = aioredis.from_url(
            get_settings().redis_url,
            decode_responses=True,
        )
    return _redis


async def close_redis() -> None:
    """关闭 Redis 连接（应用退出时调用）。"""
    global _redis
    if _redis is not None:
        try:
            await _redis.aclose()
        except Exception as e:
            logger.warning("关闭 Redis 连接失败: %s", e)
        finally:
            _redis = None
```

#### 4.3.2 提取 `runtime/task_runner.py` — 公共辅助函数（非统一执行流程）

**不**统一三个差异极大的图执行流程，而是提取真正可复用的小函数：

```python
"""Graph 执行公共辅助 — 不统一执行流程，只提取可复用的状态校验和消息发送。"""

from __future__ import annotations

from src.runtime.graph_app import get_graph_app


async def get_graph_state_values(thread_id: str) -> tuple[dict, list[str]]:
    """获取 graph 状态的 values 和 next_nodes。

    Returns: (values, next_nodes)
    """
    app = await get_graph_app()
    config = {"configurable": {"thread_id": thread_id}}
    state = await app.aget_state(config)
    values = state.values if state and state.values else {}
    next_nodes = list(state.next) if state and state.next else []
    return values, next_nodes


def is_node_in_next(node_name: str, next_nodes: list[str]) -> bool:
    return node_name in next_nodes
```

**使用场景：**
- `diagnosis_service.py`、`solution_service.py`、`review_service.py` 中大量重复的 `app = await get_graph_app(); state = await app.aget_state(config); values = state.values...` 可简化
